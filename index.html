  <script type="module">

    import { initializeApp } from "https://www.gstatic.com/firebasejs/11.0.1/firebase-app.js";

    import {

      getAuth,

      onAuthStateChanged,

      signInWithEmailAndPassword,

      signOut

    } from "https://www.gstatic.com/firebasejs/11.0.1/firebase-auth.js";

    import {

      getFirestore,

      doc,

      setDoc,

      onSnapshot

    } from "https://www.gstatic.com/firebasejs/11.0.1/firebase-firestore.js";


    const firebaseConfig = {

      apiKey: "AIzaSyBjAgIXbfKwApYcP0OMcXKp06B1ZC6mjjk",

      authDomain: "personal-todo-sync-1baef.firebaseapp.com",

      projectId: "personal-todo-sync-1baef",

      storageBucket: "personal-todo-sync-1baef.firebasestorage.app",

      messagingSenderId: "1049871009058",

      appId: "1:1049871009058:web:81b209b4892cecb60bd17b",

      measurementId: "G-8WGSTZ2G1Y"

    };


    const app = initializeApp(firebaseConfig);

    const auth = getAuth(app);

    const db = getFirestore(app);


    const sections = ['office', 'home', 'me', 'study'];

    let currentUser = null;

    let unsubscribeSnapshot = null;

    let isApplyingRemoteUpdate = false;

    let saveTimer = null;


    const loginView = document.getElementById('login-view');

    const appView = document.getElementById('app-view');

    const loginEmail = document.getElementById('login-email');

    const loginPassword = document.getElementById('login-password');

    const loginBtn = document.getElementById('login-btn');

    const loginError = document.getElementById('login-error');

    const logoutBtn = document.getElementById('logout-btn');

    const syncStatus = document.getElementById('sync-status');

    const dateInput = document.getElementById('dateInput');


    function setStatus(text) {

      syncStatus.textContent = text;

    }


    function createTaskRow(task = { text: '', completed: false }) {

      const li = document.createElement('li');

      if (task.completed) li.classList.add('completed');


      const checkbox = document.createElement('input');

      checkbox.type = 'checkbox';

      checkbox.checked = Boolean(task.completed);

      checkbox.onchange = handleCheck;


      const input = document.createElement('input');

      input.type = 'text';

      input.placeholder = 'Enter your task';

      input.value = task.text || '';

      input.oninput = () => scheduleSave();


      li.appendChild(checkbox);

      li.appendChild(input);

      return li;

    }


    window.addTask = function addTask(section, skipSave = false) {

      const ul = document.querySelector(`#${section} ul`);

      ul.appendChild(createTaskRow());

      if (!skipSave) saveTasks();

    };


    window.handleCheck = function handleCheck(e) {

      const li = e.target.closest('li');

      li.classList.toggle('completed');

      if (e.target.checked) {

        confetti({

          particleCount: 100,

          spread: 70,

          origin: { y: 0.6 }

        });

      }

      saveTasks();

    };


    window.clearSelected = function clearSelected(section) {

      const ul = document.querySelector(`#${section} ul`);

      ul.querySelectorAll('li').forEach(li => {

        if (li.querySelector('input[type="checkbox"]').checked) {

          li.remove();

        }

      });

      saveTasks();

    };


    window.resetTasks = function resetTasks() {

      sections.forEach(section => {

        const ul = document.querySelector(`#${section} ul`);

        ul.innerHTML = '';

        for (let i = 0; i < 5; i++) addTask(section, true);

      });

      saveTasks();

    };


    function collectCurrentState() {

      const data = {};

      sections.forEach(section => {

        const tasks = [];

        document.querySelectorAll(`#${section} ul li`).forEach(li => {

          tasks.push({

            text: li.querySelector('input[type="text"]').value,

            completed: li.classList.contains('completed')

          });

        });

        data[section] = tasks;

      });

      return {

        date: dateInput.value,

        sections: data,

        updatedAt: Date.now()

      };

    }


    function renderState(payload) {

      isApplyingRemoteUpdate = true;

      try {

        if (payload?.date) {

          dateInput.value = payload.date;

        } else if (!dateInput.value) {

          dateInput.value = new Date().toISOString().split('T')[0];

        }


        const sectionData = payload?.sections || {};

        sections.forEach(section => {

          const ul = document.querySelector(`#${section} ul`);

          ul.innerHTML = '';

          const items = sectionData[section] || [];

          items.forEach(task => ul.appendChild(createTaskRow(task)));

          for (let i = items.length; i < 5; i++) {

            addTask(section, true);

          }

        });

      } finally {

        isApplyingRemoteUpdate = false;

      }

    }


    function scheduleSave() {

      if (isApplyingRemoteUpdate) return;

      const state = collectCurrentState();

      localStorage.setItem('todoData', JSON.stringify(state));

      setStatus('Saving...');


      clearTimeout(saveTimer);

      saveTimer = setTimeout(() => {

        pushToCloud(state);

      }, 400);

    }


    async function pushToCloud(state) {

      if (!currentUser) return;

      try {

        const userDocRef = doc(db, 'users', currentUser.uid, 'todo', 'current');

        await setDoc(userDocRef, state, { merge: true });

        setStatus('Synced');

      } catch (err) {

        console.error(err);

        setStatus('Offline / Retry');

      }

    }


    window.saveTasks = function saveTasks() {

      scheduleSave();

    };


    dateInput.addEventListener('change', () => scheduleSave());


    loginBtn.addEventListener('click', async () => {

      loginError.textContent = '';

      try {

        await signInWithEmailAndPassword(auth, loginEmail.value.trim(), loginPassword.value);

      } catch (err) {

        loginError.textContent = 'Sign-in failed. Check your email and password.';

      }

    });


    logoutBtn.addEventListener('click', async () => {

      if (unsubscribeSnapshot) {

        unsubscribeSnapshot();

        unsubscribeSnapshot = null;

      }

      await signOut(auth);

    });


    onAuthStateChanged(auth, (user) => {

      currentUser = user;

      if (unsubscribeSnapshot) {

        unsubscribeSnapshot();

        unsubscribeSnapshot = null;

      }


      if (!user) {

        loginView.style.display = 'flex';

        appView.style.display = 'none';

        return;

      }


      loginView.style.display = 'none';

      appView.style.display = 'block';


      const cached = JSON.parse(localStorage.getItem('todoData') || 'null');

      if (cached) {

        renderState(cached);

      } else {

        renderState({ date: new Date().toISOString().split('T')[0], sections: {} });

      }


      const userDocRef = doc(db, 'users', user.uid, 'todo', 'current');

      setStatus('Connecting...');


      unsubscribeSnapshot = onSnapshot(userDocRef, (snapshot) => {

        if (snapshot.exists()) {

          const remoteData = snapshot.data();

          const localData = JSON.parse(localStorage.getItem('todoData') || 'null');

          if (!localData || (remoteData.updatedAt || 0) >= (localData.updatedAt || 0)) {

            localStorage.setItem('todoData', JSON.stringify(remoteData));

            renderState(remoteData);

          }

          setStatus('Synced');

        } else {

          const initial = collectCurrentState();

          pushToCloud(initial);

        }

      }, (err) => {

        console.error(err);

        setStatus('Sync error');

      });

    });
